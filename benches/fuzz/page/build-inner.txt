set -e
F="-g -O1 -fno-omit-frame-pointer -fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=undefined -std=gnu11 -D_GNU_SOURCE -Wno-unused-parameter -I. -I/albero/src"
for s in stubs_log fuzz_page; do clang $F -c $s.c -o /out/obj/$s.o; done
clang $F -c /albero/src/rcp.c -o /out/obj/rcp.o
# rcp.c is linked for rcp_key_address(); anything else it calls that the harness does not have becomes a stub that TRAPS (generated)
L="-lz -lpthread -lm"
clang -fsanitize=fuzzer,address,undefined /out/obj/fuzz_page.o /out/obj/rcp.o /out/obj/stubs_log.o $L -o /out/bin/page 2> /out/link1.log || true
{
	cat <<'EOS'
/* GENERATED by build.sh from the linker's list of undefined symbols: each one TRAPS if it is reached. */
#include <stdio.h>
static void __attribute__((noreturn)) hit(const char *n) { fprintf(stderr, "STUB: page/rcp reached %s, which the harness does not link\n", n); __builtin_trap(); }
EOS
	grep -o "undefined reference to \`[A-Za-z_0-9]*'" /out/link1.log | sed "s/.*\`//; s/'//" | sort -u | while read -r n; do
		echo "void $n(void); void $n(void) { hit(\"$n\"); }"
	done
} > /out/stubs_missing.c
clang $F -c /out/stubs_missing.c -o /out/obj/stubs_missing.o
clang -fsanitize=fuzzer,address,undefined /out/obj/fuzz_page.o /out/obj/rcp.o /out/obj/stubs_log.o /out/obj/stubs_missing.o $L -o /out/bin/page
python3 -I gen_seeds.py /out/seeds
cp page.dict /out/page.dict
