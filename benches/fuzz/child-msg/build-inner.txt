set -e
PK="gio-2.0 libpipewire-0.3 libdrm libva libva-drm vulkan opus libei-1.0 xkbcommon"
F="-g -O1 -fno-omit-frame-pointer -fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=undefined -std=gnu11 -D_GNU_SOURCE -Wno-unused-parameter -I. -I/albero/src $(pkg-config --cflags $PK)"
L="$(pkg-config --libs gio-2.0) -lpam -lpthread -ldl -lm"
for s in stubs_log fuzz_child_msg; do clang $F -c $s.c -o /out/obj/$s.o; done
# the product modules child.c calls are NOT linked: every undefined symbol becomes a stub that traps (stubs_missing.c, generated)
clang -fsanitize=fuzzer,address,undefined /out/obj/fuzz_child_msg.o /out/obj/stubs_log.o $L -o /out/bin/child-msg 2> /out/link1.log || true
{
	cat <<'EOS'
/* GENERATED by build.sh from the linker's list of undefined symbols: each one TRAPS if the parser reaches it. */
#include <stdio.h>
static void __attribute__((noreturn)) hit(const char *n) { fprintf(stderr, "STUB: child.c reached %s, which the harness does not link\n", n); __builtin_trap(); }
EOS
	grep -o "undefined reference to \`[A-Za-z_0-9]*'" /out/link1.log | sed "s/.*\`//; s/'//" | sort -u | while read -r n; do
		echo "void $n(void); void $n(void) { hit(\"$n\"); }"
	done
} > /out/stubs_missing.c
clang $F -c /out/stubs_missing.c -o /out/obj/stubs_missing.o
clang -fsanitize=fuzzer,address,undefined /out/obj/fuzz_child_msg.o /out/obj/stubs_log.o /out/obj/stubs_missing.o $L -o /out/bin/child-msg
python3 -I gen_seeds.py /out/seeds
cp child-msg.dict /out/child-msg.dict
